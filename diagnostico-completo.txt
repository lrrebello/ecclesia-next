"use client";
import { useState } from "react";

export default function LoginPage() {
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [message, setMessage] = useState("");

  async function submit(event: React.FormEvent) {
    event.preventDefault();
    const response = await fetch(`${process.env.NEXT_PUBLIC_API_URL}/api/v1/auth/login`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ email, password }),
    });
    const data = await response.json();
    setMessage(response.ok ? "Login realizado." : data.detail || "Erro no login.");
  }

  return (
    <main className="min-h-screen bg-slate-950 text-white flex items-center justify-center p-6">
      <form onSubmit={submit} className="w-full max-w-sm space-y-4 rounded-xl bg-slate-900 p-6">
        <h1 className="text-2xl font-bold">Entrar</h1>
        <input className="w-full rounded-lg bg-slate-800 p-3" placeholder="E-mail" value={email} onChange={(e) => setEmail(e.target.value)} />
        <input className="w-full rounded-lg bg-slate-800 p-3" type="password" placeholder="Senha" value={password} onChange={(e) => setPassword(e.target.value)} />
        <button className="w-full rounded-lg bg-indigo-500 p-3 font-medium">Entrar</button>
        {message && <p className="text-sm text-slate-300">{message}</p>}
      </form>
    </main>
  );
}

from fastapi import APIRouter, Depends, HTTPException
from pydantic import BaseModel, EmailStr
from sqlalchemy.orm import Session
from app.db.base import get_db
from app.core.models import User, Church, AuditLog
from app.security.password import hash_password, verify_password
from app.security.tokens import create_access_token, create_refresh_token
import uuid

router = APIRouter()

class RegisterInput(BaseModel):
    church_slug: str
    name: str
    email: EmailStr
    password: str
    data_consent: bool

class LoginInput(BaseModel):
    email: EmailStr
    password: str

@router.get("/health")
def health():
    return {"status": "ok"}

@router.post("/auth/register")
def register(payload: RegisterInput, db: Session = Depends(get_db)):
    if not payload.data_consent:
        raise HTTPException(400, "Consentimento de dados é obrigatório")
    church = db.query(Church).filter_by(slug=payload.church_slug).first()
    if not church:
        raise HTTPException(404, "Igreja não encontrada")
    if db.query(User).filter_by(email=payload.email.lower()).first():
        raise HTTPException(409, "E-mail já cadastrado")
    user = User(
        church_id=church.id,
        name=payload.name.strip(),
        email=payload.email.lower(),
        password_hash=hash_password(payload.password),
        status="pending",
        email_verification_token=str(uuid.uuid4()),
        data_consent_at=None,
    )
    db.add(user)
    db.commit()
    db.refresh(user)
    return {"id": user.id, "status": user.status, "email_verification_token": user.email_verification_token}

@router.post("/auth/login")
def login(payload: LoginInput, db: Session = Depends(get_db)):
    user = db.query(User).filter_by(email=payload.email.lower()).first()
    if not user or not user.password_hash or not verify_password(payload.password, user.password_hash):
        raise HTTPException(401, "E-mail ou senha inválidos")
    if user.status != "active":
        raise HTTPException(403, "Conta pendente, rejeitada ou e-mail não verificado")
    return {
        "access_token": create_access_token(user.id, user.church_id),
        "refresh_token": create_refresh_token(user.id),
        "user": {"id": user.id, "name": user.name, "email": user.email}
    }

import uuid
from datetime import datetime
from sqlalchemy import String, Boolean, DateTime, ForeignKey, Integer, Text, Numeric, JSON
from sqlalchemy.dialects.postgresql import UUID
from sqlalchemy.orm import Mapped, mapped_column, relationship
from app.db.base import Base

def uuid_default():
    return str(uuid.uuid4())

class Church(Base):
    __tablename__ = "churches"
    id: Mapped[str] = mapped_column(String(36), primary_key=True, default=uuid_default)
    name: Mapped[str] = mapped_column(String(160), nullable=False)
    slug: Mapped[str] = mapped_column(String(80), unique=True, nullable=False)
    country: Mapped[str | None] = mapped_column(String(80))
    currency_code: Mapped[str] = mapped_column(String(3), default="EUR")
    is_active: Mapped[bool] = mapped_column(Boolean, default=True)
    created_at: Mapped[datetime] = mapped_column(DateTime, default=datetime.utcnow)
    users = relationship("User", back_populates="church")

class Role(Base):
    __tablename__ = "roles"
    id: Mapped[str] = mapped_column(String(36), primary_key=True, default=uuid_default)
    church_id: Mapped[str | None] = mapped_column(ForeignKey("churches.id"), nullable=True)
    name: Mapped[str] = mapped_column(String(80), nullable=False)
    description: Mapped[str | None] = mapped_column(Text)
    is_system: Mapped[bool] = mapped_column(Boolean, default=False)
    users = relationship("UserRole", back_populates="role")

class Permission(Base):
    __tablename__ = "permissions"
    id: Mapped[str] = mapped_column(String(36), primary_key=True, default=uuid_default)
    code: Mapped[str] = mapped_column(String(120), unique=True, nullable=False)
    description: Mapped[str | None] = mapped_column(Text)

class RolePermission(Base):
    __tablename__ = "role_permissions"
    role_id: Mapped[str] = mapped_column(ForeignKey("roles.id"), primary_key=True)
    permission_id: Mapped[str] = mapped_column(ForeignKey("permissions.id"), primary_key=True)

class User(Base):
    __tablename__ = "users"
    id: Mapped[str] = mapped_column(String(36), primary_key=True, default=uuid_default)
    church_id: Mapped[str | None] = mapped_column(ForeignKey("churches.id"))
    email: Mapped[str] = mapped_column(String(255), unique=True, nullable=False, index=True)
    name: Mapped[str] = mapped_column(String(160), nullable=False)
    password_hash: Mapped[str | None] = mapped_column(String(255))
    status: Mapped[str] = mapped_column(String(20), default="pending")
    is_email_verified: Mapped[bool] = mapped_column(Boolean, default=False)
    email_verification_token: Mapped[str | None] = mapped_column(String(255), unique=True)
    reset_password_token: Mapped[str | None] = mapped_column(String(255), unique=True)
    reset_password_expires_at: Mapped[datetime | None] = mapped_column(DateTime)
    data_consent_at: Mapped[datetime | None] = mapped_column(DateTime)
    marketing_consent: Mapped[bool] = mapped_column(Boolean, default=False)

----- backend/app/security/authorization.py -----
from fastapi import Depends, HTTPException, status
from app.db.base import get_db
from app.core.models import User, UserRole, RolePermission, Permission
from sqlalchemy.orm import Session

def current_user(db: Session = Depends(get_db)) -> User | None:
    return None

def authorize(permission_code: str, scope_type: str | None = None, scope_id: str | None = None):
    def dependency(user=Depends(current_user), db: Session = Depends(get_db)):
        if user is None:
            raise HTTPException(status.HTTP_401_UNAUTHORIZED, "Não autenticado")
        query = (
            db.query(Permission.code)
            .join(RolePermission, RolePermission.permission_id == Permission.id)
            .join(UserRole, UserRole.role_id == RolePermission.role_id)
            .filter(UserRole.user_id == user.id, Permission.code == permission_code)
        )
        if scope_type:
            query = query.filter(UserRole.scope_type == scope_type)
        if scope_id:
            query = query.filter(UserRole.scope_id == scope_id)
        if not query.first():
            raise HTTPException(status.HTTP_403_FORBIDDEN, "Acesso negado")
        return True
    return dependency

pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")

def hash_password(password: str) -> str:
    return pwd_context.hash(password)

def verify_password(password: str, password_hash: str) -> bool:
    return pwd_context.verify(password, password_hash)

import uuid
from datetime import datetime, timedelta, timezone
from jose import jwt
from app.core.config import get_settings

settings = get_settings()

def create_access_token(user_id: str, church_id: str | None = None) -> str:
    expire = datetime.now(timezone.utc) + timedelta(minutes=settings.access_token_expire_minutes)
    payload = {"sub": user_id, "church_id": church_id, "exp": expire, "type": "access"}
    return jwt.encode(payload, settings.secret_key, algorithm="HS256")

def create_refresh_token(user_id: str) -> str:
    expire = datetime.now(timezone.utc) + timedelta(days=settings.refresh_token_expire_days)
    payload = {"sub": user_id, "exp": expire, "type": "refresh", "jti": str(uuid.uuid4())}
    return jwt.encode(payload, settings.secret_key, algorithm="HS256")

def decode_token(token: str) -> dict:
    return jwt.decode(token, settings.secret_key, algorithms=["HS256"])

from app.core.config import get_settings

class AIService:
    def __init__(self):
        self.settings = get_settings()

    def generate_quiz(self, topic: str, quantity: int = 5, difficulty: str = "medium") -> dict:
        prompt = f"""
Crie {quantity} questões de múltipla escolha sobre: {topic}.
Dificuldade: {difficulty}.
Responda apenas em JSON válido com esta estrutura:
{{"questions":[{{"question":"","options":["","","",""],"correct_index":0,"explanation":""}}]}}
"""
        if self.settings.ai_provider == "openai":
            return self._openai(prompt)
        if self.settings.ai_provider == "gemini":
            return self._gemini(prompt)
        raise ValueError("Provedor de IA não configurado")

    def _openai(self, prompt: str) -> dict:
        from openai import OpenAI
        client = OpenAI(api_key=self.settings.openai_api_key)
        response = client.chat.completions.create(
            model=self.settings.openai_model,
            messages=[{"role": "user", "content": prompt}],
            response_format={"type": "json_object"},
        )
        import json
        return json.loads(response.choices[0].message.content)

    def _gemini(self, prompt: str) -> dict:
        from google import genai
        client = genai.Client(api_key=self.settings.google_ai_api_key)
        response = client.models.generate_content(
            model=self.settings.gemini_model,
            contents=prompt,
        )
        import json, re
        text = response.text.strip().removeprefix("```json").removesuffix("```").strip()
        return json.loads(text)

----- backend/alembic/env.py -----
from logging.config import fileConfig
from sqlalchemy import engine_from_config, pool
from alembic import context
from app.core.config import get_settings
from app.db.base import Base
from app.core import models

config = context.config
if config.config_file_name:
    fileConfig(config.config_file_name)
config.set_main_option("sqlalchemy.url", get_settings().database_url)
target_metadata = Base.metadata

def run_migrations_offline():
    context.configure(url=config.get_main_option("sqlalchemy.url"), target_metadata=target_metadata, literal_binds=True)
    with context.begin_transaction():
        context.run_migrations()

def run_migrations_online():
    connectable = engine_from_config(config.get_section(config.config_ini_section), prefix="sqlalchemy.", poolclass=pool.NullPool)
    with connectable.connect() as connection:
        context.configure(connection=connection, target_metadata=target_metadata)
        with context.begin_transaction():
            context.run_migrations()

if context.is_offline_mode():
    run_migrations_offline()
else:
    run_migrations_online()
